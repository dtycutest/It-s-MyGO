## ⚡ 必做的前5项工作

### 1️⃣ 创建API模块 (15分钟)

复制以下代码到 `src/api/index.js`：

```javascript
// HTTP请求基础配置
import axios from 'axios'

const API_BASE_URL = process.env.VUE_APP_API_URL || 'https://api.ycg.com'
const request = axios.create({
  baseURL: API_BASE_URL + '/api/v1',
  timeout: 30000
})

// 请求拦截器
request.interceptors.request.use(config => {
  const token = localStorage.getItem('token')
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

// 响应拦截器
request.interceptors.response.use(
  response => response.data,
  error => Promise.reject(error)
)

export default request
```

创建 `src/api/product.js`：

```javascript
import request from './index'

// 搜索商品
export function searchProducts(keyword, params) {
  return request.get('/products/search', {
    params: { keyword, ...params }
  })
}

// 获取商品详情
export function getProductDetail(productId) {
  return request.get(`/products/${productId}`)
}

// 获取热搜榜
export function getTrendingKeywords() {
  return request.get('/products/trends')
}
```

### 3️⃣ 配置状态管理 (10分钟)

创建 `src/store/index.js`：

```javascript
import { createPinia } from 'pinia'

const pinia = createPinia()

export default pinia
```

创建 `src/store/user.js`：

```javascript
import { defineStore } from 'pinia'
import { ref } from 'vue'

export const useUserStore = defineStore('user', () => {
  const token = ref(localStorage.getItem('token') || '')
  const userInfo = ref(null)

  const setToken = (newToken) => {
    token.value = newToken
    localStorage.setItem('token', newToken)
  }

  const setUserInfo = (info) => {
    userInfo.value = info
  }

  const logout = () => {
    token.value = ''
    userInfo.value = null
    localStorage.removeItem('token')
  }

  return {
    token,
    userInfo,
    setToken,
    setUserInfo,
    logout
  }
})
```

### 5️⃣ 创建首页 (20分钟)

创建 `src/pages/index/index.vue`：

```vue
<template>
  <div class="index-page">
    <!-- 搜索栏 -->
    <div class="search-bar">
      <input 
        v-model="searchKeyword"
        type="text"
        placeholder="搜索商品"
        @keyup.enter="handleSearch"
      >
      <button @click="handleSearch">搜索</button>
    </div>

    <!-- 分类导航 -->
    <div class="categories">
      <div v-for="cat in categories" :key="cat" class="category-item">
        {{ cat }}
      </div>
    </div>

    <!-- 热搜榜 -->
    <div class="trending-section">
      <h3>🔥 热搜榜</h3>
      <div v-for="trend in trends" :key="trend" class="trend-item">
        {{ trend }}
      </div>
    </div>

    <!-- 推荐商品 -->
    <div class="products-section">
      <h3>推荐商品</h3>
      <div class="products-grid">
        <product-card
          v-for="product in products"
          :key="product.id"
          :product="product"
          @click="goToDetail(product.id)"
        />
      </div>
    </div>
  </div>
</template>

<script setup>
import { ref, onMounted } from 'vue'
import { usePinia } from 'pinia'
import { searchProducts, getTrendingKeywords } from '@/api/product'
import ProductCard from '@/components/ProductCard.vue'

const searchKeyword = ref('')
const categories = ref(['电子', '服装', '家居', '食品', '美妆', '运动'])
const trends = ref(['iPhone 15', 'MacBook Pro', '小米 14', 'iPad Pro', 'AirPods'])
const products = ref([])

onMounted(() => {
  loadTrends()
  loadProducts()
})

async function loadTrends() {
  try {
    const response = await getTrendingKeywords()
    trends.value = response.data.map(t => t.keyword)
  } catch (error) {
    console.error('加载热搜失败:', error)
  }
}

async function loadProducts() {
  try {
    const response = await searchProducts('手机', { pageNum: 1, pageSize: 10 })
    products.value = response.data.list
  } catch (error) {
    console.error('加载商品失败:', error)
  }
}

async function handleSearch() {
  if (!searchKeyword.value.trim()) return
  
  try {
    const response = await searchProducts(searchKeyword.value)
    products.value = response.data.list
  } catch (error) {
    console.error('搜索失败:', error)
  }
}

function goToDetail(productId) {
  uni.navigateTo({
    url: `/pages/productDetail/productDetail?id=${productId}`
  })
}
</script>
```

## 🐛 常见问题

### Q: 如何切换不同的API环境？
**A:** 修改 `.env.development` 或 `.env.production` 中的 `VUE_APP_API_URL`

### Q: 如何在本地调试小程序？
**A:** 使用微信开发者工具导入项目，或在HBuilderX中点击"运行"→"在微信开发者工具中打开"

### Q: 组件如何传参和事件通信？
**A:** 参考 ProductCard.vue 的示例，使用 `defineProps` 和 `emit` 或 `@click`

### Q: 如何处理API请求错误？
**A:** 在 `request.js` 的响应拦截器中统一处理，或在具体页面中 try-catch
